# Host tests for the layout generator: no game, no SDK. doctest is vendored in third_party/doctest.
add_executable(layout_tests layout_tests.cpp ../src/layout/layout.cpp)
target_include_directories(layout_tests PRIVATE ../src/layout ../third_party/doctest)
if(MSVC)
	target_compile_options(layout_tests PRIVATE /W4 /permissive- /utf-8)
else()
	target_compile_options(layout_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME layout_tests COMMAND layout_tests)

# Host tests for the outdoor piece-picker. The Jungle catalogue rows it tests against are in fixtures/.
add_executable(outdoor_tests outdoor_tests.cpp ../src/outdoor/outdoor.cpp ../src/layout/layout.cpp)
target_include_directories(outdoor_tests PRIVATE ../src/outdoor ../src/layout ../third_party/doctest)
target_compile_definitions(outdoor_tests PRIVATE CARTOGRAPHER_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/fixtures")
if(MSVC)
	target_compile_options(outdoor_tests PRIVATE /W4 /permissive- /utf-8)
else()
	target_compile_options(outdoor_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME outdoor_tests COMMAND outdoor_tests)

cmake_minimum_required(VERSION 3.29)
project(armory LANGUAGES CXX RC)

set(ARMORY_VERSION "0.0.1" CACHE STRING "Experimental Armory plugin version")
set(ARMORY_SDK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/PluginSDK" CACHE PATH "D2RLoader PluginSDK directory")
option(ARMORY_TESTS "Build the portable Armory tests" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(D2RLPLUGIN_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory("${ARMORY_SDK_DIR}" PluginSDK EXCLUDE_FROM_ALL)

add_library(armory SHARED src/plugin.cpp src/log.cpp src/armory.rc)
target_link_libraries(armory PRIVATE D2RLPlugin::D2RLPlugin)
target_compile_definitions(armory PRIVATE ARMORY_VERSION="${ARMORY_VERSION}")
set_target_properties(armory PROPERTIES PREFIX "")
if(MSVC)
    target_compile_options(armory PRIVATE /W4 /permissive- /EHsc /utf-8)
    set_property(TARGET armory PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
endif()
if(ARMORY_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()

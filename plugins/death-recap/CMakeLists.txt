cmake_minimum_required(VERSION 3.29)
project(death_recap LANGUAGES CXX RC)

set(D2RL_SDK_DIR "" CACHE PATH "D2RLoader PluginSDK directory")
if(NOT D2RL_SDK_DIR)
	message(FATAL_ERROR "Pass -DD2RL_SDK_DIR=<PluginSDK checkout>")
endif()

set(D2RLPLUGIN_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory(${D2RL_SDK_DIR} PluginSDK EXCLUDE_FROM_ALL)

add_library(death_recap SHARED src/plugin.cpp src/native.cpp src/panel.cpp src/names.cpp src/history.cpp src/log.cpp src/death-recap.rc)
target_compile_features(death_recap PRIVATE cxx_std_20)
target_link_libraries(death_recap PRIVATE D2RLPlugin::D2RLPlugin user32 comctl32)
set_target_properties(death_recap PROPERTIES OUTPUT_NAME "death-recap")
d2rlplugin_embed_config(TARGET death_recap FILE death-recap.toml)
if(MSVC)
	target_compile_options(death_recap PRIVATE /W4 /permissive- /EHsc /utf-8)
	set_property(TARGET death_recap PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
endif()

enable_testing()
add_executable(death_recap_tests tests/telemetry_test.cpp)
target_compile_features(death_recap_tests PRIVATE cxx_std_20)
target_include_directories(death_recap_tests PRIVATE src)
add_test(NAME death_recap_tests COMMAND death_recap_tests)
add_executable(death_recap_adapter_tests tests/adapter_test.cpp src/history.cpp)
target_compile_features(death_recap_adapter_tests PRIVATE cxx_std_20)
target_include_directories(death_recap_adapter_tests PRIVATE src)
if(MSVC)
	target_compile_options(death_recap_adapter_tests PRIVATE /W4 /permissive- /EHsc /utf-8)
	set_property(TARGET death_recap_adapter_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
endif()
add_test(NAME death_recap_adapter_tests COMMAND death_recap_adapter_tests)
add_executable(death_recap_native_tests tests/native_test.cpp src/log.cpp)
target_compile_features(death_recap_native_tests PRIVATE cxx_std_20)
target_include_directories(death_recap_native_tests PRIVATE src)
target_link_libraries(death_recap_native_tests PRIVATE D2RLPlugin::D2RLPlugin user32)
if(MSVC)
	target_compile_options(death_recap_native_tests PRIVATE /W4 /permissive- /EHsc /utf-8)
	set_property(TARGET death_recap_native_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
endif()
add_test(NAME death_recap_native_tests COMMAND death_recap_native_tests)

add_executable(death_recap_view_tests tests/view_test.cpp)
target_compile_features(death_recap_view_tests PRIVATE cxx_std_20)
target_include_directories(death_recap_view_tests PRIVATE src)
if(MSVC)
    target_compile_options(death_recap_view_tests PRIVATE /W4 /permissive- /EHsc /utf-8)
    set_property(TARGET death_recap_view_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
endif()
add_test(NAME death_recap_view_tests COMMAND death_recap_view_tests)
